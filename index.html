FINAL FIRESTORE RULES Firebase Console → Firestore Database → Rules → পুরোনো Rules সম্পূর্ণ replace করে এটি Publish করবে। rules_version = '2'; service cloud.firestore { match /databases/{database}/documents { /* ========================= AUTHENTICATION ========================= */ function signedIn() { return request.auth != null; } function isOwner(uid) { return signedIn() && request.auth.uid == uid; } /* ========================= USERS ========================= */ match /users/{uid} { // Logged-in users can read profiles. allow read: if signedIn(); // User can create only own profile. allow create: if signedIn() && request.auth.uid == uid && request.resource.data.uid == request.auth.uid; // User can update only own profile. allow update: if signedIn() && request.auth.uid == uid && request.resource.data.uid == resource.data.uid; // User can delete only own profile. allow delete: if signedIn() && request.auth.uid == uid; } /* ========================= CHATS ========================= */ match /chats/{chatId} { // Only participants can read chat. allow read: if signedIn() && request.auth.uid in resource.data.participants; // Create exactly two-person chat. allow create: if signedIn() && request.resource.data.participants is list && request.resource.data.participants.size() == 2 && request.auth.uid in request.resource.data.participants; // Only existing participants can update. allow update: if signedIn() && request.auth.uid in resource.data.participants && request.resource.data.participants == resource.data.participants; // Do not delete entire chat. allow delete: if false; /* ========================= MESSAGES ========================= */ match /messages/{messageId} { // Only chat participants can read. allow read: if signedIn() && request.auth.uid in get( /databases/$(database)/documents/chats/$(chatId) ).data.participants; // Send message. allow create: if signedIn() // Sender must be current user. && request.resource.data.senderId == request.auth.uid // Receiver must be string. && request.resource.data.receiverId is string // Cannot send to yourself. && request.resource.data.receiverId != request.auth.uid // Sender must belong to chat. && request.auth.uid in get( /databases/$(database)/documents/chats/$(chatId) ).data.participants // Receiver must belong to same chat. && request.resource.data.receiverId in get( /databases/$(database)/documents/chats/$(chatId) ).data.participants // Text validation. && request.resource.data.text is string && request.resource.data.text.size() > 0 && request.resource.data.text.size() <= 2000; // Sender can delete own message. allow delete: if signedIn() && resource.data.senderId == request.auth.uid; // Messages cannot be edited. allow update: if false; } } /* ========================= EVERYTHING ELSE ========================= */ match /{document=**} { allow read, write: if false;
